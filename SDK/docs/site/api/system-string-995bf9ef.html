Type/API example

string commandLine = "echo hello";
int split = commandLine.IndexOf(' ');
string command = commandLine.Substring(0, split);
string arguments = commandLine.Substring(split + 1);
Console.WriteLine(String.Concat(command, ": ", arguments));
